# Everything the installer can select. Read by setup.sh and setup.ps1.
#
# Format: group|item|label|platforms|default|action
#   platforms  comma list of linux, macos, wsl, windows (WSL also counts as linux)
#   default    on or off, used until a saved selection says otherwise
#   action     pkg                   package, name mapped by pkg_name in setup.sh
#              scoop:<bucket>/<app>  Windows app installed by Scoop
#              step:<name>           named install step; items may share a step
#
# Group names and item names form the keys in ~/.local/bootstrap/config, so
# renaming one loses that item's saved choice.
